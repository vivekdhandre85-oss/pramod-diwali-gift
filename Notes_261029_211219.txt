<script>
const STORAGE_KEY = "pe_qr_gift_demo_v1";
let records = {};
let currentCode = "";
let currentGift = "";
let giftOptions = [
"Bluetooth Earphones",
"Smart Watch",
"Shopping Voucher",
"Electronics Accessory",
"Special Surprise Gift"
];

function readRecords(){
try { records = JSON.parse(localStorage.getItem(STORAGE_KEY) || "{}"); }
catch(e){ records = {}; }
}
function saveRecords(){
localStorage.setItem(STORAGE_KEY, JSON.stringify(records));
renderRecords();
}
function showPage(page){
["customer","owner","qrpage"].forEach(id=>{
document.getElementById(id).classList.toggle("hidden",id!page);
});
if(page="owner") renderRecords();
}
function saveSettings(){
const title=document.getElementById("cfgTitle").value.trim();
const message=document.getElementById("cfgMessage").value.trim();
const list=document.getElementById("giftList").value.split("\n").map(x=>x.trim()).filter(Boolean);
if(!list.length){alert("Please enter at least one gift.");return;}
giftOptions=list;
document.getElementById("offerTitle").textContent=title || "Pramod Electronics";
document.getElementById("offerMessage").textContent=message;
alert("Demo settings updated in this page.");
}
document.getElementById("videoFile").addEventListener("change",function(){
const file=this.files && this.files[0];
if(!file)return;
const video=document.getElementById("shopVideo");
if(video.dataset.objectUrl) URL.revokeObjectURL(video.dataset.objectUrl);
const url=URL.createObjectURL(file);
video.dataset.objectUrl=url;
video.src=url;
video.muted=true;
video.play().catch(()=>{});
});
function getGift(){
if(!giftOptions.length) giftOptions=["Surprise Gift"];
// DEMO ONLY: browser-generated code, not secure or globally unique.
let code;
do {
code="PE-GIFT-"+Math.random().toString(36).slice(2,8).toUpperCase();
} while(records[code]);
const gift=giftOptions[Math.floor(Math.random()*giftOptions.length)];
records[code]={
code:code,
gift:gift,
issued:new Date().toLocaleString(),
redeemed:false,
redeemedAt:""
};
currentCode=code;
currentGift=gift;
saveRecords();
document.getElementById("giftCode").textContent=code;
document.getElementById("giftName").textContent=gift;
document.getElementById("giftResult").classList.remove("hidden");
}
function copyGift(){
if(!currentCode)return;
if(navigator.clipboard && navigator.clipboard.writeText){
navigator.clipboard.writeText(currentCode).then(()=>alert("Code copied!")).catch(()=>prompt("Copy this code:",currentCode));
} else prompt("Copy this code:",currentCode);
}
function verifyCode(){
const code=document.getElementById("verifyInput").value.trim().toUpperCase();
const rec=records[code];
const box=document.getElementById("verifyResult");
if(!rec){box.innerHTML="<p style='color:#c21f35'>Code not found in this browser's demo register.</p>";return;}
box.textContent="";
const title=document.createElement("h3");
title.textContent=rec.redeemed?"Already redeemed":"Valid demo code";
title.style.color=rec.redeemed?"#c21f35":"#087443";
const details=document.createElement("p");
details.textContent="Gift: "+rec.gift+" | Issued: "+rec.issued+(rec.redeemed?" | Redeemed: "+rec.redeemedAt:"");
box.append(title,details);
}
function redeemCode(){
const code=document.getElementById("verifyInput").value.trim().toUpperCase();
const rec=records[code];
const box=document.getElementById("verifyResult");
if(!rec){box.textContent="Code not found.";return;}
if(rec.redeemed){box.textContent="This demo code has already been redeemed.";return;}
rec.redeemed=true;
rec.redeemedAt=new Date().toLocaleString();
saveRecords();
box.textContent="Demo code marked redeemed. Gift: "+rec.gift;
}
function renderRecords(){
const list=document.getElementById("codeList");
if(!list)return;
list.textContent="";
const items=Object.values(records).reverse();
document.getElementById("totalCodes").textContent=items.length;
document.getElementById("usedCodes").textContent=items.filter(x=>x.redeemed).length;
if(!items.length){list.textContent="No demo codes yet.";return;}
items.forEach(rec=>{
const div=document.createElement("div");
div.className="item";
const strong=document.createElement("strong");
strong.textContent=rec.code+" — "+rec.gift;
const info=document.createElement("div");
info.className="muted";
info.textContent="Issued: "+rec.issued+" • "+(rec.redeemed?"Redeemed":"Not redeemed");
div.append(strong,info);
list.append(div);
});
}
function exportCSV(){
const rows=[["Code","Gift","Issued","Redeemed","Redeemed At"]];
Object.values(records).forEach(r=>rows.push([r.code,r.gift,r.issued,r.redeemed?"Yes":"No",r.redeemedAt]));
const csv=rows.map(row=>row.map(v=>'"'+String(v||"").replace(/"/g,'""')+'"').join(",")).join("\r\n");
const blob=new Blob([csv],{type:"text/csv;charset=utf-8;"});
const a=document.createElement("a");
a.href=URL.createObjectURL(blob);
a.download="pramod-demo-gift-codes.csv";
a.click();
URL.revokeObjectURL(a.href);
}
function clearDemo(){
if(!confirm("Clear all demo gift codes from this browser?"))return;
records={};
saveRecords();
document.getElementById("verifyResult").textContent="";
}
function makeQR(){
const url=document.getElementById("siteURL").value.trim();
const out=document.getElementById("qr");
out.textContent="";
if(!/^https:///i.test(url)){
out.textContent="Please enter a public HTTPS website URL.";
return;
}
if(!window.QRCode){
out.textContent="QR library is loading or blocked. Connect to the internet and reload the page.";
const s=document.createElement("script");
s.src="https://cdnjs.cloudflare.com/ajax/libs/qrcodejs/1.0.0/qrcode.min.js";
s.onload=()=>{out.textContent="";new QRCode(out,{text:url,width:220,height:220});};
document.head.appendChild(s);
return;
}
new QRCode(out,{text:url,width:220,height:220});
}
readRecords();
renderRecords();
document.getElementById("offerTitle").textContent=document.getElementById("cfgTitle").value;
document.getElementById("offerMessage").textContent=document.getElementById("cfgMessage").value;
</script>
